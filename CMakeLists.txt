# Co-op's own art -- the Luca plant and its cheese -- is compiled into the
# executable, so the mod still runs from a stock game folder with nothing extra
# to copy beside it. Each assets/coop/<name>.png becomes gCoopArt_<name>[] and
# gCoopArt_<name>_size; editing a PNG re-runs this on the next build.
set(COOP_ART_NAMES luca cheese roman yeti_spray cursor_fill cursor_outline kernel_squash philippe_pepper)
# Art that may not exist yet: embedded when the PNG is there, an empty
# placeholder (size 0, which the game treats as "not drawn") when it is not.
set(COOP_ART_OPTIONAL adventure_plus_background)
set(COOP_ART_SOURCE "${PROJECT_BINARY_DIR}/CoopArtData.cpp")
set(aCoopArtText "// Generated by CMakeLists.txt from assets/coop/*.png. Do not edit.\n#include <cstddef>\n")
foreach(aName IN LISTS COOP_ART_NAMES)
	set(aPath "${CMAKE_CURRENT_SOURCE_DIR}/assets/coop/${aName}.png")
	set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${aPath}")
	file(READ "${aPath}" aHex HEX)
	string(LENGTH "${aHex}" aHexLength)
	math(EXPR aSize "${aHexLength} / 2")
	string(REGEX REPLACE "(([0-9a-f][0-9a-f]){32})" "\\1\n" aHex "${aHex}")
	string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," aBytes "${aHex}")
	string(APPEND aCoopArtText "extern const unsigned char gCoopArt_${aName}[] = {\n${aBytes}\n};\nextern const std::size_t gCoopArt_${aName}_size = ${aSize};\n")
endforeach()
foreach(aName IN LISTS COOP_ART_OPTIONAL)
	set(aPath "${CMAKE_CURRENT_SOURCE_DIR}/assets/coop/${aName}.png")
	set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/coop")
	if(EXISTS "${aPath}")
		set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${aPath}")
		file(READ "${aPath}" aHex HEX)
		string(LENGTH "${aHex}" aHexLength)
		math(EXPR aSize "${aHexLength} / 2")
		string(REGEX REPLACE "(([0-9a-f][0-9a-f]){32})" "\\1\n" aHex "${aHex}")
		string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," aBytes "${aHex}")
		string(APPEND aCoopArtText "extern const unsigned char gCoopArt_${aName}[] = {\n${aBytes}\n};\nextern const std::size_t gCoopArt_${aName}_size = ${aSize};\n")
	else()
		string(APPEND aCoopArtText "extern const unsigned char gCoopArt_${aName}[] = { 0 };\nextern const std::size_t gCoopArt_${aName}_size = 0;\n")
	endif()
endforeach()
# Adventure Plus zombie parts: repainted copies of the game's own zombie
# parts, each named exactly like the part it replaces (see
# art-templates/zombies/README.txt). Any number, including none.
file(GLOB COOP_ZOMBIE_PARTS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/coop/zombie-parts/*.png")
list(SORT COOP_ZOMBIE_PARTS)
set(aPartTable "")
set(aPartIndex 0)
foreach(aPath IN LISTS COOP_ZOMBIE_PARTS)
	get_filename_component(aStem "${aPath}" NAME_WE)
	file(READ "${aPath}" aHex HEX)
	string(LENGTH "${aHex}" aHexLength)
	math(EXPR aSize "${aHexLength} / 2")
	string(REGEX REPLACE "(([0-9a-f][0-9a-f]){32})" "\\1\n" aHex "${aHex}")
	string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," aBytes "${aHex}")
	string(APPEND aCoopArtText "static const unsigned char gCoopZombiePart${aPartIndex}[] = {\n${aBytes}\n};\n")
	string(APPEND aPartTable "\t{ \"${aStem}\", gCoopZombiePart${aPartIndex}, ${aSize} },\n")
	math(EXPR aPartIndex "${aPartIndex} + 1")
endforeach()
string(APPEND aCoopArtText "struct CoopArtPart { const char* mName; const unsigned char* mData; std::size_t mSize; };\n")
string(APPEND aCoopArtText "extern const CoopArtPart gCoopZombieParts[] = {\n${aPartTable}\t{ nullptr, nullptr, 0 }\n};\n")
string(APPEND aCoopArtText "extern const int gCoopZombiePartCount = ${aPartIndex};\n")
# Through a temporary and a copy, so an unchanged PNG does not touch the
# generated file and force a recompile.
file(WRITE "${COOP_ART_SOURCE}.tmp" "${aCoopArtText}")
configure_file("${COOP_ART_SOURCE}.tmp" "${COOP_ART_SOURCE}" COPYONLY)
list(APPEND SOURCES "${COOP_ART_SOURCE}")

function(_pvz_patch_sdl2_iface_libs target)
	get_target_property(_libs ${target} INTERFACE_LINK_LIBRARIES)
	if(NOT _libs)
		return()
	endif()
	set(_patched)
	foreach(_lib IN LISTS _libs)
		if(_lib STREQUAL "SDL2::SDL2")
			list(APPEND _patched SDL2::SDL2-static)
		else()
			list(APPEND _patched "${_lib}")
		endif()
	endforeach()
	set_target_properties(${target} PROPERTIES INTERFACE_LINK_LIBRARIES "${_patched}")
endfunction()
